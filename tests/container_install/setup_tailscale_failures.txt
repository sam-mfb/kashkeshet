== Tailscale does not start
unset    COMPOSE_ENV_FILES COMPOSE_FILE COMPOSE_PATH_SEPARATOR COMPOSE_PROFILES COMPOSE_PROJECT_NAME KKT_PORT KKT_VERSION TS_AUTHKEY TS_HOSTNAME TS_TAILNET
run      docker compose version
mkdir    /data/docker
mkdir    /data/docker/backups
mkdir    /data/docker
write    plain /data/docker/compose.tailscale.yaml
  = the repository's compose.tailscale.yaml
mkdir    /data/docker/tailscale
write    plain /data/docker/tailscale/serve.json
  = the repository's tailscale/serve.json
write    private /data/docker/.env
  | # Written by `kkt setup`. Setup writes it again when you run it or upgrade.
  | KKT_VERSION=0.4.0
  | COMPOSE_PROJECT_NAME=kkt-test
  | KKT_PORT=7421
  | KKT_ALLOW_HOSTS=kkt.example,10.0.0.5
  | COMPOSE_FILE=compose.tailscale.yaml
  | TS_HOSTNAME=kkt
  | TS_TAILNET=pending.invalid
attached cwd=/data/docker  docker compose --project-directory /data/docker run --rm --no-deps --user root --entrypoint chown kkt 10001 /backups
say      Starting Tailscale...
attached cwd=/data/docker  docker compose --project-directory /data/docker up -d tailscale
error    `docker compose --project-directory /data/docker up -d tailscale` failed (exit 1: no such image)
